\begin{frame}{One content property, multiple metrics}
\footnotesize
\renewcommand{\arraystretch}{1.04}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{.19\linewidth}>{\raggedright\arraybackslash}p{.76\linewidth}@{}}
\toprule
Level & Content\\
\midrule
Perspective &
\PerspectiveTag{\ContentIcon}{Content}\hspace{.4em} explanation claim about the model/data relationship\\
Property &
\textbf{Correctness / faithfulness}: does $E(f,x)$ track the behavior of $f$ around $x$?\\
Metric families &
\textbf{Perturbation response}: deletion, insertion, AOPC\\[-.2em]
& \textbf{Attribution agreement}: faithfulness correlation, infidelity\\[-.2em]
& \textbf{Sanity controls}: randomization, synthetic / white-box checks\\
Protocol choices &
\textbf{Perturbation}: baseline, masking rule, feature grouping\\[-.2em]
& \textbf{Measurement}: normalization, number of samples, random seed\\
\bottomrule
\end{tabular}
\renewcommand{\arraystretch}{1}

\vspace{.55em}
\scriptsize\raggedright Same property, different operationalizations: each metric exposes different assumptions and failure modes.\cite{quantus}\par
\end{frame}

\begin{frame}{Properties can conflict}
\centering\Large
\PerspectiveTag{\ContentIcon}{Content}: a faithful explanation may be verbose.\\[.6em]
\PerspectiveTag{\PresentationIcon}{Presentation}: a compact explanation may omit relevant behavior.\\[.6em]
\normalsize The evaluation objective must state which trade-offs are acceptable.
